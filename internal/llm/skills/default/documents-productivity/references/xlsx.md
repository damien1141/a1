# XLSX (Excel/Sheets) — openpyxl, pandas, xlsxwriter, CSV

Create, edit, analyze, and convert spreadsheets. `openpyxl` 3.1+ for cells/formulas/styles/charts; `pandas` for bulk data; `xlsxwriter` for advanced charts/formatting; `csv` module for CSV/TSV. Preserve leading zeros, dates, and IDs.

## Create with `openpyxl` 3.1+

```python
from openpyxl import Workbook
from openpyxl.styles import Font, PatternFill, Border, Side, Alignment, NamedStyle
from openpyxl.chart import BarChart, LineChart, PieChart, Reference
from openpyxl.formatting.rule import ColorScaleRule, CellIsRule, FormulaRule
from openpyxl.worksheet.datavalidation import DataValidation
from openpyxl.utils import get_column_letter

wb = Workbook()
ws = wb.active
ws.title = 'Sales'

# --- Styling primitives ---
header_fill = PatternFill('solid', fgColor='0D9488')
header_font = Font(bold=True, color='FFFFFF', size=11)
thin = Side(style='thin', color='CCCCCC')
border = Border(left=thin, right=thin, top=thin, bottom=thin)

# --- Header row ---
headers = ['Region', 'Q2 ARR', 'Q3 ARR', 'Growth', 'Growth %']
for col, h in enumerate(headers, start=1):
    c = ws.cell(row=1, column=col, value=h)
    c.fill = header_fill
    c.font = header_font
    c.alignment = Alignment(horizontal='center')
    c.border = border

# --- Data rows with formulas ---
data = [
    ('NA',   1_940_000, 2_100_000),
    ('EMEA', 1_150_000, 1_400_000),
    ('APAC',   620_000,   700_000),
]
for i, (region, q2, q3) in enumerate(data, start=2):
    ws.cell(row=i, column=1, value=region).border = border
    ws.cell(row=i, column=2, value=q2).number_format = '$#,##0'
    ws.cell(row=i, column=3, value=q3).number_format = '$#,##0'
    ws.cell(row=i, column=4, value=f'=C{i}-B{i}').number_format = '$#,##0'
    ws.cell(row=i, column=5, value=f'=(C{i}-B{i})/B{i}').number_format = '0.0%'

# --- Total row ---
total_row = len(data) + 2
ws.cell(row=total_row, column=1, value='Total').font = Font(bold=True)
ws.cell(row=total_row, column=2, value=f'=SUM(B2:B{total_row-1})').number_format = '$#,##0'
ws.cell(row=total_row, column=3, value=f'=SUM(C2:C{total_row-1})').number_format = '$#,##0'
ws.cell(row=total_row, column=4, value=f'=SUM(D2:D{total_row-1})').number_format = '$#,##0'
ws.cell(row=total_row, column=5, value=f'=(C{total_row}-B{total_row})/B{total_row}').number_format = '0.0%'

# --- Freeze panes, column widths ---
ws.freeze_panes = 'A2'   # keeps row 1 visible
for col, width in enumerate([12, 14, 14, 14, 12], start=1):
    ws.column_dimensions[get_column_letter(col)].width = width

# --- Conditional formatting (color scale on Growth %) ---
ws.conditional_formatting.add(
    f'E2:E{total_row-1}',
    ColorScaleRule(
        start_type='min', start_color='F87171',     # red
        mid_type='percentile', mid_value=50, mid_color='FBBF24',  # yellow
        end_type='max', end_color='34D399',          # green
    ),
)
# Highlight negative growth red
ws.conditional_formatting.add(
    f'E2:E{total_row-1}',
    CellIsRule(operator='lessThan', formula=['0'], fill=PatternFill('solid', fgColor='FEE2E2')),
)

# --- Data validation (dropdown for Region on future rows) ---
dv = DataValidation(type='list', formula1='"NA,EMEA,APAC,LATAM"', allow_blank=True)
dv.error = 'Pick a region from the list.'
dv.errorTitle = 'Invalid region'
ws.add_data_validation(dv)
dv.add(f'A{total_row+1}:A100')

# --- Chart ---
chart = BarChart()
chart.title = 'Q2 vs Q3 ARR by Region'
chart.type = 'col'
chart.y_axis.title = 'ARR ($)'
chart.x_axis.title = 'Region'
values = Reference(ws, min_col=2, max_col=3, min_row=1, max_row=total_row - 1)
cats = Reference(ws, min_col=1, min_row=2, max_row=total_row - 1)
chart.add_data(values, titles_from_data=True)
chart.set_categories(cats)
chart.width = 18
chart.height = 10
ws.add_chart(chart, 'G2')

# --- Named range (for use in formulas across sheets) ---
from openpyxl.workbook.defined_name import DefinedName
wb.defined_names['TotalQ3'] = DefinedName(name='TotalQ3', attr_text=f"'Sales'!$C${total_row}")

# --- Second sheet ---
ws2 = wb.create_sheet('Notes')
ws2['A1'] = 'Generated by documents-productivity skill'
ws2['A1'].font = Font(italic=True, color='6B7280')

wb.save('output/sales_report.xlsx')
```

## Read Existing XLSX

```python
from openpyxl import load_workbook

# load_workbook — reads everything (slow for large files)
wb = load_workbook('input/existing.xlsx', data_only=False)  # data_only=True → cached values, not formulas
print(f'Sheets: {wb.sheetnames}')

for sheet_name in wb.sheetnames:
    ws = wb[sheet_name]
    print(f'\n=== {sheet_name} ===')
    print(f'Dimensions: {ws.dimensions}, max_row={ws.max_row}, max_col={ws.max_column}')
    print(f'Frozen panes: {ws.freeze_panes}')
    for row in ws.iter_rows(min_row=1, max_row=min(5, ws.max_row), values_only=False):
        for cell in row:
            print(f'  {cell.coordinate}: value={cell.value!r}, fmt={cell.number_format!r}')

# Read formulas vs values:
# - data_only=False (default) → formulas as strings
# - data_only=True → cached values (ONLY if Excel saved them; openpyxl doesn't calculate)
```

## pandas — bulk read/write

```python
import pandas as pd

# Read all sheets
xls = pd.ExcelFile('input/existing.xlsx')
print(f'Sheets: {xls.sheet_names}')

# Read a single sheet — IDs as strings to preserve leading zeros
df = pd.read_excel('input/existing.xlsx', sheet_name='Sales', dtype={'SKU': str, 'ZipCode': str})
print(df.head())
print(df.dtypes)

# Multi-index pivot
pivot = df.pivot_table(index='Region', columns='Quarter', values='ARR', aggfunc='sum')
print(pivot)

# Write to Excel with formatting
with pd.ExcelWriter('output/sales_pivot.xlsx', engine='openpyxl') as writer:
    df.to_excel(writer, sheet_name='Raw', index=False)
    pivot.to_excel(writer, sheet_name='Pivot')

    # Access the workbook for post-processing
    wb = writer.book
    ws = writer.sheets['Raw']
    ws.column_dimensions['A'].width = 20
```

## Large Sheet Performance

| Approach | Use Case | Speed |
|---|---|---|
| `openpyxl.load_workbook` | Read-modify-write small/medium files (<50k rows) | Slow |
| `openpyxl.Workbook(write_only=True)` | Write huge files (no read) | Fast |
| `xlsxwriter` | Write huge files with advanced formatting | Fastest |
| `pandas.read_excel(engine='openpyxl')` | Read for analysis | Medium |
| `pandas.read_excel(engine='calamine')` | Read very large files (10x faster than openpyxl) | Very fast |
| `csv` module | CSV/TSV — no formatting needed | Fastest |

```python
# Write-only mode for huge files
from openpyxl import Workbook
wb = Workbook(write_only=True)
ws = wb.create_sheet('BigData')
# ws.append() only — no cell access
for i in range(1_000_000):
    ws.append([i, f'item-{i}', i * 1.5])
wb.save('output/big.xlsx')
```

```python
# xlsxwriter for advanced formatting
import xlsxwriter
wb = xlsxwriter.Workbook('output/advanced.xlsx')
ws = wb.add_worksheet('Sales')

fmt_header = wb.add_format({'bold': True, 'bg_color': '#0D9488', 'font_color': 'white', 'align': 'center', 'border': 1})
fmt_money = wb.add_format({'num_format': '$#,##0', 'border': 1})
fmt_pct = wb.add_format({'num_format': '0.0%', 'border': 1})

ws.write_row('A1', ['Region', 'Q2', 'Q3', 'Growth %'], fmt_header)
ws.write_row('A2', ['NA', 1940000, 2100000, None], fmt_money)  # A2 fmt applies to first cell only
ws.write('B2', 1940000, fmt_money)
ws.write('C2', 2100000, fmt_money)
ws.write('D2', '=(C2-B2)/B2', fmt_pct)

# Chart with xlsxwriter (richer formatting than openpyxl)
chart = wb.add_chart({'type': 'column'})
chart.add_series({
    'name': 'Q2',
    'categories': ['Sales', 1, 0, 4, 0],
    'values':     ['Sales', 1, 1, 4, 1],
    'fill': {'color': '#0D9488'},
})
ws.insert_chart('G2', chart)
wb.close()
```

## CSV / TSV

```python
import csv

# Read CSV
with open('input/data.csv', newline='', encoding='utf-8') as f:
    reader = csv.DictReader(f)
    for row in reader:
        print(row['Region'], row['ARR'])

# Write CSV
with open('output/data.csv', 'w', newline='', encoding='utf-8') as f:
    writer = csv.DictWriter(f, fieldnames=['Region', 'ARR', 'Growth'])
    writer.writeheader()
    for row in [('NA', 2_100_000, 0.08), ('EMEA', 1_400_000, 0.22)]:
        writer.writerow({'Region': row[0], 'ARR': row[1], 'Growth': row[2]})

# TSV: use delimiter='\t'
# csv.register_dialect('tsv', delimiter='\t', quoting=csv.QUOTE_MINIMAL)

# pandas for CSV (faster for large files)
import pandas as pd
df = pd.read_csv('input/data.csv', dtype={'SKU': str, 'ZipCode': str})  # preserve leading zeros
df.to_csv('output/data.csv', index=False)
```

## Conditional Formatting

```python
from openpyxl.formatting.rule import ColorScaleRule, CellIsRule, FormulaRule, IconSetRule
from openpyxl.styles import PatternFill

# Color scale (heatmap)
ws.conditional_formatting.add('B2:B100',
    ColorScaleRule(
        start_type='min', start_color='F87171',
        mid_type='percentile', mid_value=50, mid_color='FBBF24',
        end_type='max', end_color='34D399',
    ))

# CellIs — operator-based
ws.conditional_formatting.add('C2:C100',
    CellIsRule(operator='greaterThan', formula=['1000000'],
               fill=PatternFill('solid', fgColor='D1FAE5')))

# FormulaRule — custom formula
ws.conditional_formatting.add('A2:Z100',
    FormulaRule(formula=['MOD(ROW(),2)=0'],
                fill=PatternFill('solid', fgColor='F3F4F6')))  # zebra stripes

# Icon set
from openpyxl.formatting.rule import IconSetRule
ws.conditional_formatting.add('D2:D100', IconSetRule('3Arrows', 'percent', [0, 33, 67]))
```

## Pivot Tables

`openpyxl` reads pivot tables but cannot create them from scratch. For programmatic pivots:

1. Build the pivot in `pandas`, write the result with `to_excel`.
2. Or write the source data, then have the user insert a pivot in Excel.
3. Or use `xlsxwriter` (also limited — no native pivot creation; can write a pivot cache).

```python
import pandas as pd
df = pd.read_excel('input/sales.xlsx')
pivot = df.pivot_table(index='Region', columns='Quarter', values='ARR', aggfunc='sum', margins=True)
pivot.to_excel('output/sales_pivot.xlsx', sheet_name='Pivot')
```

## Verification

```python
from openpyxl import load_workbook
import subprocess

wb = load_workbook('output/sales_report.xlsx')
print(f'Sheets: {wb.sheetnames}')
for name in wb.sheetnames:
    ws = wb[name]
    print(f'  {name}: dims={ws.dimensions}, max_row={ws.max_row}, max_col={ws.max_column}')

# Spot-check a formula cell
ws = wb['Sales']
print(f'C2 formula: {ws["C2"].value}')        # value (or formula)
print(f'E2 formula: {ws["E2"].value}')        # formula string

# Integrity check
result = subprocess.run(['unzip', '-t', 'output/sales_report.xlsx'], capture_output=True, text=True)
print('ZIP OK' if result.returncode == 0 else 'CORRUPT')
```

## Common Gotchas

- **`data_only=True` returns `None` for formulas not previously calculated by Excel** — openpyxl does not evaluate formulas. For calculated values, open with LibreOffice first or use `xlwings`/`formulas` library.
- **Leading zeros on IDs** — `dtype={'SKU': str}` in pandas; in openpyxl, set `cell.number_format = '@'` (text format) BEFORE writing the value.
- **Dates** — `openpyxl` returns `datetime.datetime` objects; pandas returns `Timestamp`. Both serialize correctly; just be explicit on read with `parse_dates=[...]`.
- **`max_row`/`max_column` can lie** — they reflect the bounding box of cells with values or styles. A stray empty cell with formatting can inflate them.
- **Conditional formatting rules accumulate** — calling `ws.conditional_formatting.add` repeatedly on the same range stacks rules. Clear with `ws.conditional_formatting = ConditionalFormattingList()` (rare).
- **Charts don't render in LibreOffice headless the same as Excel** — bar/column/line are safe; complex chart types (sunburst, treemap) may not render. Test with the target viewer.
- **`xlsxwriter` cannot READ** — it's write-only. Use `openpyxl` for read-modify-write.
- **CSV is not a spreadsheet** — no formulas, no formatting, no multiple sheets. Use XLSX for those features.
- **Encoding** — always specify `encoding='utf-8'` on CSV open. Excel on Windows defaults to cp1252.
